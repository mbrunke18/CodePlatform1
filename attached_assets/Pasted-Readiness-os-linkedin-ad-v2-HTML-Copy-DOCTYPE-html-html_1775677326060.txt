function anim(el, style, dur = 500) {
  return new Promise(r => {
    if (typeof el === 'string') el = document.getElementById(el);
    if (!el) { r(); return; }
    el.style.transition = `all ${dur}ms cubic-bezier(0.16, 1, 0.3, 1)`;

    const t = setTimeout(r, dur);

  });
}
