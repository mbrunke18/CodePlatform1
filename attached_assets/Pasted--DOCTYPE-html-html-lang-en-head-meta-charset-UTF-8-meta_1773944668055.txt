  // DateTime
  function updateTime() {
    const now = new Date();
    const opts = { weekday: 'short', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit' };
    document.getElementById('datetime').textContent = now.toLocaleDateString('en-US', opts).toUpperCase();
  }
  updateTime();
  setInterval(updateTime, 1000);

  // Last scan
  let scanSeconds = 0;
  setInterval(() => {
    scanSeconds++;
    const el = document.getElementById('lastscan');
    if (scanSeconds < 60) el.textContent = scanSeconds + 'S AGO';
    else el.textContent = Math.floor(scanSeconds/60) + 'M AGO';
  }, 1000);

  // Radar canvas
  const canvas = document.getElementById('radar');
  const ctx = canvas.getContext('2d');
  const cx = 90, cy = 90, maxR = 82;

  // Blips
  const blips = Array.from({length: 12}, () => ({
    angle: Math.random() * Math.PI * 2,
    r: 20 + Math.random() * 58,
    speed: (Math.random() - 0.5) * 0.004,
    size: 2 + Math.random() * 2.5,
    alpha: 0.4 + Math.random() * 0.6,
    color: Math.random() > 0.6 ? '#C9A84C' : '#2B8A6E',
    pulse: Math.random() * Math.PI * 2
  }));

  // Sweep angle
  let sweep = 0;

  function drawRadar() {
    ctx.clearRect(0, 0, 180, 180);

    // Rings
    for (let i = 1; i <= 4; i++) {
      ctx.beginPath();
      ctx.arc(cx, cy, (maxR / 4) * i, 0, Math.PI * 2);
      ctx.strokeStyle = 'rgba(43,138,110,0.15)';
      ctx.lineWidth = 1;
      ctx.stroke();
    }

    // Cross hairs
    ctx.strokeStyle = 'rgba(43,138,110,0.1)';
    ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(cx, cy - maxR); ctx.lineTo(cx, cy + maxR); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(cx - maxR, cy); ctx.lineTo(cx + maxR, cy); ctx.stroke();

    // Sweep gradient
    const sweepGrad = ctx.createConicalGradient
      ? null
      : null;

    // Sweep arc
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(sweep);

    const grad = ctx.createLinearGradient(0, 0, maxR, 0);
    grad.addColorStop(0, 'rgba(43,138,110,0.5)');
    grad.addColorStop(1, 'rgba(43,138,110,0)');

    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.arc(0, 0, maxR, -0.4, 0.4);
    ctx.closePath();
    ctx.fillStyle = 'rgba(43,138,110,0.12)';
    ctx.fill();

    // Sweep line
    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.lineTo(maxR, 0);
    ctx.strokeStyle = 'rgba(43,138,110,0.7)';
    ctx.lineWidth = 1.5;
    ctx.stroke();

    ctx.restore();

    // Blips
    blips.forEach(b => {
      b.angle += b.speed;
      b.pulse += 0.05;
      const x = cx + Math.cos(b.angle) * b.r;
      const y = cy + Math.sin(b.angle) * b.r;
      const alpha = b.alpha * (0.7 + 0.3 * Math.sin(b.pulse));

      ctx.beginPath();
      ctx.arc(x, y, b.size, 0, Math.PI * 2);
      ctx.fillStyle = b.color + Math.floor(alpha * 255).toString(16).padStart(2, '0');
      ctx.fill();

      // Glow
      ctx.beginPath();
      ctx.arc(x, y, b.size * 2.5, 0, Math.PI * 2);
      ctx.fillStyle = b.color + '15';
      ctx.fill();
    });

    // Center dot
    ctx.beginPath();
    ctx.arc(cx, cy, 3, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(43,138,110,0.8)';
    ctx.fill();

    sweep += 0.018;
    requestAnimationFrame(drawRadar);
  }

  drawRadar();
</script>
