/**
 * Mock Salesforce Service
 * Simulates Salesforce REST API for demo purposes
 */

import { MOCK_DEALS, mockDealToSalesforce, calculateDealRiskScore, isDeaLatRisk, getRiskLevel } from '../seeds/dealRiskDemoData';
import type { MockDeal, SalesforceOpportunity } from '../seeds/dealRiskDemoData';

export class MockSalesforceService {
  private deals: Map<string, MockDeal> = new Map();
  private accessToken?: string;
  private lastQueried: Date = new Date();

  constructor() {
    MOCK_DEALS.forEach(deal => {
      this.deals.set(deal.id, { ...deal });
    });
  }

  async authenticate(clientId: string, clientSecret: string): Promise<string> {
    console.log(`[Mock Salesforce] Authenticating with client: ${clientId}`);
    await new Promise(resolve => setTimeout(resolve, 500));
    this.accessToken = `mock_token_${Date.now()}`;
    return this.accessToken;
  }

  async queryOpportunities(): Promise<SalesforceOpportunity[]> {
    this.lastQueried = new Date();
    return Array.from(this.deals.values()).map(mockDealToSalesforce);
  }

  async queryOpportunitiesWithFilters(filters: {
    minAmount?: number;
    stage?: string;
    riskScoreMin?: number;
  }): Promise<SalesforceOpportunity[]> {
    return Array.from(this.deals.values())
      .filter(deal => {
        if (filters.minAmount && deal.amount < filters.minAmount) return false;
        if (filters.stage && deal.stage !== filters.stage) return false;
        if (filters.riskScoreMin) {
          const riskScore = calculateDealRiskScore(deal);
          if (riskScore < filters.riskScoreMin) return false;
        }
        return true;
      })
      .map(mockDealToSalesforce);
  }

  async getDealsAtRisk(): Promise<Array<SalesforceOpportunity & { riskScore: number }>> {
    return Array.from(this.deals.values())
      .filter(isDeaLatRisk)
      .map(deal => ({
        ...mockDealToSalesforce(deal),
        riskScore: calculateDealRiskScore(deal),
      }));
  }

  async getDealById(dealId: string): Promise<(SalesforceOpportunity & { riskScore: number; riskLevel: string }) | null> {
    const deal = this.deals.get(dealId);
    if (!deal) return null;

    const riskScore = calculateDealRiskScore(deal);
    return {
      ...mockDealToSalesforce(deal),
      riskScore,
      riskLevel: getRiskLevel(riskScore),
    };
  }

  getPipelineSummary(): {
    totalDeals: number;
    totalPipeline: number;
    dealsAtRisk: number;
    criticalRisk: number;
    averageWinProbability: number;
    lastQueried: Date;
  } {
    const dealsArray = Array.from(this.deals.values());
    const atRiskDeals = dealsArray.filter(isDeaLatRisk);
    const criticalRiskDeals = atRiskDeals.filter(d => calculateDealRiskScore(d) >= 80);

    const totalPipeline = dealsArray.reduce((sum, deal) => sum + deal.amount, 0);
    const avgWinProb = dealsArray.reduce((sum, deal) => sum + deal.winProbability, 0) / dealsArray.length;

    return {
      totalDeals: dealsArray.length,
      totalPipeline,
      dealsAtRisk: atRiskDeals.length,
      criticalRisk: criticalRiskDeals.length,
      averageWinProbability: Math.round(avgWinProb),
      lastQueried: this.lastQueried,
    };
  }

  updateDeal(dealId: string, updates: Partial<MockDeal>): MockDeal | null {
    const deal = this.deals.get(dealId);
    if (!deal) return null;

    const updated = { ...deal, ...updates, lastModified: new Date() };
    this.deals.set(dealId, updated);
    return updated;
  }

  detectTriggers(): Array<{
    dealId: string;
    dealName: string;
    amount: number;
    riskScore: number;
    triggerReason: string;
    recommendedPlaybook: string;
  }> {
    const triggeredDeals: ReturnType<typeof this.detectTriggers> = [];

    Array.from(this.deals.values()).forEach(deal => {
      if (isDeaLatRisk(deal)) {
        const riskScore = calculateDealRiskScore(deal);
        let triggerReason = '';
        
        if (deal.riskIndicators.probability > 25) {
          triggerReason = `Win probability dropped ${deal.riskIndicators.probability}%`;
        } else if (deal.riskIndicators.closeDateCompression > 14) {
          triggerReason = `Close date compressed by ${deal.riskIndicators.closeDateCompression} days`;
        }

        triggeredDeals.push({
          dealId: deal.id,
          dealName: deal.name,
          amount: deal.amount,
          riskScore,
          triggerReason,
          recommendedPlaybook: 'Deal Risk Response',
        });
      }
    });

    return triggeredDeals;
  }

  reset(): void {
    this.deals.clear();
    MOCK_DEALS.forEach(deal => {
      this.deals.set(deal.id, { ...deal });
    });
    console.log('[Mock Salesforce] Reset to initial state');
  }
}

/**
 * DETECT Phase Service - Background monitoring for deal risk triggers
 */

import { mockSalesforce } from './MockSalesforceService';

export interface TriggerAlert {
  triggerId: string;
  dealId: string;
  dealName: string;
  amount: number;
  riskScore: number;
  confidence: number;
  triggerReason: string;
  detectedAt: Date;
  recommendedPlaybook: string;
  urgency: 'low' | 'medium' | 'high' | 'critical';
}

export class TriggerDetectionService {
  private isMonitoring = false;
  private monitoringInterval: NodeJS.Timeout | null = null;
  private lastDetectionTime = new Date();
  private alertCallbacks: ((alert: TriggerAlert) => void)[] = [];

  startMonitoring(interval: number = 15) {
    if (this.isMonitoring) {
      console.log('[DETECT] Already monitoring');
      return;
    }

    this.isMonitoring = true;
    console.log(`[DETECT] Starting trigger detection service (${interval}s interval)`);

    this.monitoringInterval = setInterval(() => {
      this.checkForTriggers();
    }, interval * 1000);

    this.checkForTriggers();
  }

  stopMonitoring() {
    if (this.monitoringInterval) {
      clearInterval(this.monitoringInterval);
      this.monitoringInterval = null;
    }
    this.isMonitoring = false;
    console.log('[DETECT] Stopped trigger detection service');
  }

  private async checkForTriggers() {
    try {
      const triggers = mockSalesforce.detectTriggers();
      
      if (triggers.length > 0) {
        console.log(`[DETECT] Found ${triggers.length} trigger conditions met`);
        
        triggers.forEach(trigger => {
          const alert: TriggerAlert = {
            triggerId: `trigger-${trigger.dealId}-${Date.now()}`,
            dealId: trigger.dealId,
            dealName: trigger.dealName,
            amount: trigger.amount,
            riskScore: trigger.riskScore,
            confidence: this.calculateConfidence(trigger.riskScore),
            triggerReason: trigger.triggerReason,
            detectedAt: new Date(),
            recommendedPlaybook: trigger.recommendedPlaybook,
            urgency: this.getUrgency(trigger.riskScore, trigger.amount),
          };

          this.emitAlert(alert);
          this.saveAlert(alert);
        });
      }

      this.lastDetectionTime = new Date();
    } catch (error) {
      console.error('[DETECT] Error checking for triggers:', error);
    }
  }

  private calculateConfidence(riskScore: number): number {
    if (riskScore < 30) return 30;
    if (riskScore < 60) return 60;
    if (riskScore < 80) return 85;
    return 95;
  }

  private getUrgency(
    riskScore: number,
    dealAmount: number
  ): 'low' | 'medium' | 'high' | 'critical' {
    if (dealAmount >= 5_000_000 && riskScore >= 80) return 'critical';
    if (dealAmount >= 3_000_000 && riskScore >= 70) return 'high';
    if (riskScore >= 60) return 'high';
    if (riskScore >= 40) return 'medium';
    return 'low';
  }

  onAlert(callback: (alert: TriggerAlert) => void) {
    this.alertCallbacks.push(callback);
  }

  private emitAlert(alert: TriggerAlert) {
    this.alertCallbacks.forEach(callback => {
      try {
        callback(alert);
      } catch (error) {
        console.error('[DETECT] Error in alert callback:', error);
      }
    });
  }

  private async saveAlert(alert: TriggerAlert) {
    try {
      console.log(`[DETECT] Alert created: ${alert.dealName} (Risk: ${alert.riskScore}%, Confidence: ${alert.confidence}%)`);
    } catch (error) {
      console.error('[DETECT] Error saving alert:', error);
    }
  }

  getStatus() {
    return {
      isMonitoring: this.isMonitoring,
      lastDetectionTime: this.lastDetectionTime,
      subscriberCount: this.alertCallbacks.length,
    };
  }

  getCurrentTriggers(): TriggerAlert[] {
    const triggers = mockSalesforce.detectTriggers();
    return triggers.map(trigger => ({
      triggerId: `trigger-${trigger.dealId}-${Date.now()}`,
      dealId: trigger.dealId,
      dealName: trigger.dealName,
      amount: trigger.amount,
      riskScore: trigger.riskScore,
      confidence: this.calculateConfidence(trigger.riskScore),
      triggerReason: trigger.triggerReason,
      detectedAt: new Date(),
      recommendedPlaybook: trigger.recommendedPlaybook,
      urgency: this.getUrgency(trigger.riskScore, trigger.amount),
    }));
  }
}

export const triggerDetectionService = new TriggerDetectionService();

export function calculateDealRiskScore(deal: MockDeal): number {
  const probabilityRisk = Math.min(deal.riskIndicators.probability * 2, 100);
  const compressionRisk = Math.min((deal.riskIndicators.closeDateCompression / 30) * 30, 30);
  const engagementRisk = deal.riskIndicators.decisionMakerEngagement === 'low' ? 15 : deal.riskIndicators.decisionMakerEngagement === 'medium' ? 8 : 0;
  const budgetRisk = deal.riskIndicators.budgetConfirmed ? 0 : 10;
  const competitorRisk = deal.riskIndicators.competitorThreat ? 15 : 0;

  const totalRisk = Math.min(probabilityRisk + compressionRisk + engagementRisk + budgetRisk + competitorRisk, 100);
  return Math.round(totalRisk);
}

export function getRiskLevel(score: number): 'low' | 'medium' | 'high' | 'critical' {
  if (score < 30) return 'low';
  if (score < 60) return 'medium';
  if (score < 80) return 'high';
  return 'critical';
}

export function isDeaLatRisk(deal: MockDeal): boolean {
  const aboveMinAmount = deal.amount >= 3_000_000;
  const significantProbDrop = deal.riskIndicators.probability > 20;
  const significantCompression = deal.riskIndicators.closeDateCompression > 14;
  
  return aboveMinAmount && (significantProbDrop || significantCompression);
}

export function getDealRiskSummary(deal: MockDeal): string {
  const reasons: string[] = [];
  
  if (deal.riskIndicators.probability > 20) {
    reasons.push(`Win probability dropped ${deal.riskIndicators.probability}%`);
  }
  if (deal.riskIndicators.closeDateCompression > 7) {
    reasons.push(`Close date compressed by ${deal.riskIndicators.closeDateCompression} days`);
  }
  if (!deal.riskIndicators.budgetConfirmed) {
    reasons.push('Budget not yet confirmed');
  }
  if (deal.riskIndicators.decisionMakerEngagement === 'low') {
    reasons.push('Low decision-maker engagement');
  }
  if (deal.riskIndicators.competitorThreat) {
    reasons.push('Competitor threat detected');
  }
  
  return reasons.join(' • ');
}
