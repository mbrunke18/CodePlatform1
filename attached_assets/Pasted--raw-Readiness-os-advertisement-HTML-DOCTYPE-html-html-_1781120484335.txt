let currentScene = -1;
let sceneTimer = null;
let progressTimer = null;
let progressStart = null;
let totalDuration = SCENES.reduce((s, sc) => s + sc.duration, 0);
let elapsed = 0;
let prevElapsed = 0;
 
function showScene(index) {
  if (currentScene >= 0 && currentScene < SCENES.length) {
    const prev = SCENES[currentScene];
    document.getElementById(prev.id).classList.remove('active');
    prev.onExit();
  }
 
  currentScene = index;
 
  if (index >= SCENES.length) {
    return;
  }
 
  const scene = SCENES[index];
  document.getElementById(scene.id).classList.add('active');
  scene.onEnter();
 
  sceneTimer = setTimeout(() => {
    showScene(index + 1);
  }, scene.duration);
}
 
function updateProgress() {
  const now = Date.now();
  if (!progressStart) return;
  elapsed = prevElapsed + (now - progressStart);
  const pct = Math.min(100, (elapsed / totalDuration) * 100);
  document.getElementById('progress').style.width = pct + '%';
  if (elapsed < totalDuration) {
    requestAnimationFrame(updateProgress);
  }
}

    requestAnimationFrame(updateProgress);

  }, 100);
}
