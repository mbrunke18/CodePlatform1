SECTION 4: REACT COMPONENT - POISE LOGO

CREATE THIS FILE: /src/components/PoiseLogo.tsx

---BEGIN PoiseLogo.tsx---

import React from 'react';

interface PoiseLogoProps {
  width?: number;
  height?: number;
  showTagline?: boolean;
  variant?: 'full' | 'icon-only' | 'text-only';
  className?: string;
}

export const PoiseLogo: React.FC<PoiseLogoProps> = ({
  width = 400,
  height = 150,
  showTagline = true,
  variant = 'full',
  className = '',
}) => {
  const aspectRatio = 800 / 200;
  const computedHeight = width ? width / aspectRatio : height;

  return (
    <svg
      viewBox="0 0 800 200"
      width={width}
      height={computedHeight}
      xmlns="http://www.w3.org/2000/svg"
      className={`poise-logo ${className}`}
      style={{ maxWidth: '100%', height: 'auto' }}
    >
      <defs>
        <style>{`
          @import url('https://fonts.googleapis.com/css2?family=Playfair+Display:wght@700&display=swap');
          .poise-text {
            font-family: 'Playfair Display', serif;
            font-size: 120px;
            font-weight: 700;
            fill: white;
            letter-spacing: 2px;
          }
          .compass-needle {
            fill: #D4AF37;
          }
          .wave-line {
            stroke: #00A8A8;
            stroke-width: 2;
            fill: none;
            stroke-linecap: round;
          }
          .tagline-text {
            font-family: Georgia, serif;
            font-size: 14px;
            fill: #D4AF37;
            letter-spacing: 1px;
          }
        `}</style>
      </defs>

      {/* Main POISE Text */}
      {variant !== 'icon-only' && (
        <text x="50" y="130" className="poise-text">
          POISE
        </text>
      )}

      {/* Compass Needle (replaces the I) */}
      {variant !== 'text-only' && (
        <g transform="translate(280, 70)">
          {/* Compass Circle Background */}
          <circle
            cx="0"
            cy="0"
            r="35"
            fill="none"
            stroke="#00A8A8"
            strokeWidth="1"
            opacity="0.3"
          />

          {/* Compass Needle - Gold Triangle pointing up */}
          <polygon
            points="0,-28 8,-8 -8,-8"
            className="compass-needle"
            opacity="0.9"
          />

          {/* Needle Base Circle */}
          <circle cx="0" cy="0" r="4" fill="#D4AF37" />
        </g>
      )}

      {/* Wave Lines beneath POISE (AI responsiveness indicator) */}
      {variant !== 'icon-only' && (
        <g transform="translate(50, 155)">
          {/* Wave 1 */}
          <path
            d="M 0 0 Q 15 -8, 30 0 T 60 0 T 90 0 T 120 0 T 150 0"
            className="wave-line"
            opacity="0.7"
          />

          {/* Wave 2 */}
          <path
            d="M 0 8 Q 15 0, 30 8 T 60 8 T 90 8 T 120 8 T 150 8"
            className="wave-line"
            opacity="0.5"
          />

          {/* Wave 3 */}
          <path
            d="M 0 16 Q 15 8, 30 16 T 60 16 T 90 16 T 120 16 T 150 16"
            className="wave-line"
            opacity="0.3"
          />
        </g>
      )}

      {/* Tagline beneath */}
      {showTagline && variant !== 'icon-only' && (
        <text x="50" y="195" className="tagline-text">
          Composure in every decision.
        </text>
      )}
    </svg>
  );
};

export default PoiseLogo;

---END PoiseLogo.tsx---

SECTION 11: QUESTIONS & TROUBLESHOOTING
================================================================================

Q: Do I need to update the database schema?
A: No. No database changes needed. M and POISE are the same product, same data structure.

Q: What if I've already launched with "M"?
A: You can still rebrand. Update your customer-facing dashboards first, then 
gradually migrate internal systems. Offer current customers the option to see 
POISE branding or keep M (during transition period).

Q: Do I rename ALL files to POISE?
A: No. Keep your file structure clean. Rename components (PoisePlaybook.tsx) and 
update copy, but don't rename your entire /src/modules structure. Keep it organized.

Q: What about the logo file?
A: Save it as poise_logo.svg or poise_logo.png in /public/logos/ or import the 
React component directly. No need to change every reference.

Q: Can I use just the compass needle icon?
A: Yes. Use variant="icon-only" on the PoiseLogo component for small headers 
and favicons.

Q: Should I trademark POISE?
A: Yes. File a Class 42 trademark application with the USPTO. This is enterprise 
software, so IP protection matters.
